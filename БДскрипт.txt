
DROP TABLE IF EXISTS "AspNetUserTokens" CASCADE;
DROP TABLE IF EXISTS "AspNetUserLogins" CASCADE;
DROP TABLE IF EXISTS "AspNetUserClaims" CASCADE;
DROP TABLE IF EXISTS "AspNetUserRoles" CASCADE;
DROP TABLE IF EXISTS "AspNetRoles" CASCADE;
DROP TABLE IF EXISTS "Bookings" CASCADE;
DROP TABLE IF EXISTS "Services" CASCADE;
DROP TABLE IF EXISTS "AspNetUsers" CASCADE;
DROP TABLE IF EXISTS "__EFMigrationsHistory" CASCADE;

CREATE TABLE "AspNetUsers" (
    "Id" TEXT PRIMARY KEY,
    "UserName" VARCHAR(256) NULL,
    "NormalizedUserName" VARCHAR(256) NULL,
    "Email" VARCHAR(256) NULL,
    "NormalizedEmail" VARCHAR(256) NULL,
    "EmailConfirmed" BOOLEAN NOT NULL DEFAULT FALSE,
    "PasswordHash" TEXT NULL,           -- Хэшированный пароль
    "SecurityStamp" TEXT NULL,
    "ConcurrencyStamp" TEXT NULL,
    "PhoneNumber" TEXT NULL,
    "PhoneNumberConfirmed" BOOLEAN NOT NULL DEFAULT FALSE,
    "TwoFactorEnabled" BOOLEAN NOT NULL DEFAULT FALSE,
    "LockoutEnd" TIMESTAMP NULL,
    "LockoutEnabled" BOOLEAN NOT NULL DEFAULT TRUE,
    "AccessFailedCount" INTEGER NOT NULL DEFAULT 0,
    -- Наши дополнительные поля
    "LastName" TEXT NOT NULL DEFAULT '',
    "FirstName" TEXT NOT NULL DEFAULT '',
    "MiddleName" TEXT NULL,
    "RegisteredAt" TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "IsAdmin" BOOLEAN NOT NULL DEFAULT FALSE
);


CREATE TABLE "Services" (
    "Id" SERIAL PRIMARY KEY,
    "Name" VARCHAR(100) NOT NULL,
    "Price" DECIMAL(10,2) NOT NULL,
    "DurationMinutes" INTEGER NOT NULL,
    "Description" VARCHAR(500) NULL,
    "IsActive" BOOLEAN NOT NULL DEFAULT TRUE
);


CREATE TABLE "Bookings" (
    "Id" SERIAL PRIMARY KEY,
    "BookingDate" DATE NOT NULL,
    "BookingTime" TIME WITHOUT TIME ZONE NOT NULL,
    "Status" VARCHAR(20) NOT NULL DEFAULT 'Pending',
    "Notes" TEXT NULL,
    "CreatedAt" TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "ServiceId" INTEGER NOT NULL,
    "UserId" TEXT NOT NULL,
    
    CONSTRAINT "FK_Bookings_Services_ServiceId" 
        FOREIGN KEY ("ServiceId") REFERENCES "Services"("Id") ON DELETE RESTRICT,
    CONSTRAINT "FK_Bookings_AspNetUsers_UserId" 
        FOREIGN KEY ("UserId") REFERENCES "AspNetUsers"("Id") ON DELETE CASCADE,
    CONSTRAINT "UC_Booking_DateTime" UNIQUE ("BookingDate", "BookingTime")
);


CREATE INDEX "IX_Bookings_BookingDate" ON "Bookings"("BookingDate");
CREATE INDEX "IX_Bookings_ServiceId" ON "Bookings"("ServiceId");
CREATE INDEX "IX_Bookings_UserId" ON "Bookings"("UserId");
CREATE INDEX "EmailIndex" ON "AspNetUsers"("NormalizedEmail");
CREATE UNIQUE INDEX "UserNameIndex" ON "AspNetUsers"("NormalizedUserName");
